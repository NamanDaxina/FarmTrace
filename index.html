Deployment checklist

Secure shared database

Apply these rules in Firebase Console → Firestore Database → Rules, then publish. The site preview cannot publish Firebase rules for you.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function role() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    function validRole() { return signedIn() && role() in ['farm', 'mandi', 'shop', 'home']; }
    match /users/{uid} {
      allow read: if signedIn() && request.auth.uid == uid;
      allow create, update, delete: if false;
    }
    match /batches/{batchId} {
      allow read: if validRole();
      allow create: if signedIn() && role() == 'farm';
      allow update: if signedIn() && role() in ['farm', 'mandi', 'shop'];
      allow delete: if signedIn() && role() == 'farm';
    }
    match /audit/{eventId} {
      allow read: if signedIn() && role() in ['farm', 'mandi', 'shop'];
      allow create: if signedIn() && role() in ['farm', 'mandi', 'shop'];
      allow update, delete: if false;
    }
  }
}

These starter rules block unauthenticated access and prevent users from editing their own role. Test role permissions before using real data. Create user role documents through the Firebase Console or a trusted admin process. Do not expose service-account keys in this website.
